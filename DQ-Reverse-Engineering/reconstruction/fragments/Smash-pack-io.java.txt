public boolean createWritableDirectory(String path, boolean deleteOnExit) {
    try {
        java.io.File directory = new java.io.File(path);
        if (!directory.exists() && !directory.mkdirs()) {
            return false;
        }
        if (deleteOnExit) {
            directory.deleteOnExit();
        }
        java.lang.SecurityManager securityManager = System.getSecurityManager();
        if (securityManager != null) {
            securityManager.checkWrite(path);
        }
        return true;
    } catch (SecurityException exception) {
        exception.printStackTrace();
        return false;
    }
}

public boolean extractAssetFromAPK(String assetName,
                                   String destinationDirectory,
                                   String outputName) {
    if (!createWritableDirectory(destinationDirectory, true)) {
        return false;
    }
    try {
        java.io.InputStream input = getAssets().open(assetName);
        if (input == null) {
            return false;
        }
        java.io.FileOutputStream output = new java.io.FileOutputStream(
            destinationDirectory + "/" + outputName
        );
        byte[] buffer = new byte[0x80000]; // 512 KiB
        while (input.available() > 0) {
            int available = input.available();
            int length = Math.min(available, buffer.length);
            input.read(buffer, 0, length);
            // Mirroring the observed bytecode: it writes the requested length,
            // not the return value from read(). This can mishandle short reads.
            output.write(buffer, 0, length);
        }
        input.close();
        output.close();
        return true;
    } catch (java.io.IOException exception) {
        exception.printStackTrace();
        return false;
    }
}
